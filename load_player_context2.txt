loadPlayer(post_id, chapter_st, sv, type) {
                let playerWrapper = jQuery('#halim-player-wrapper');
                jQuery.ajax({
                    type: 'get',
					url: 'https://hhpanda.st/player/player.php',
                    data: {
                        action: 'dox_ajax_player',
                        post_id: post_id,
                        chapter_st: chapter_st,
                        type: type,
                        sv: sv,
                    },
                    beforeSend: function () {
                        playerWrapper.find('iframe').attr('src', '');
                        playerWrapper.find('iframe').addClass('iframe-placeholder')
                    },
                    success: function (data) {
                        playerWrapper.html(data);
                    },
                });
            }
